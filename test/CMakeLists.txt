# Preserve original Release assertion policy, definitions and executable paths.
function(navigation2d_test name library)
  add_executable(${name} ${name}.cc)
  target_link_libraries(${name} PRIVATE ${library})
  set_target_properties(${name} PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/..")
  add_test(NAME ${name} COMMAND ${name} ${ARGN})
  set_property(GLOBAL APPEND PROPERTY NAVIGATION2D_TEST_TARGETS ${name})
endfunction()

function(navigation2d_assert_test name library)
  navigation2d_test(${name} ${library} ${ARGN})
  target_compile_options(${name} PRIVATE -UNDEBUG)
endfunction()
